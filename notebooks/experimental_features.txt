# --- Haar Wavelet Transform Implementation (Simplified and Potentially Misleading) ---

def _single_level_haar_transform(data_segment):
    """
    Applies a single level of Haar transform to a data segment.
    Pads with edge value if length is odd to ensure even pairs.
    Returns approximation and detail coefficients.
    """
    n = len(data_segment)
    if n < 2:
        return data_segment, np.array([]) # Cannot transform if too short

    # Pad if length is odd to ensure even pairs for Haar
    if n % 2 != 0:
        data_segment = np.pad(data_segment, (0, 1), mode='edge')
        n = len(data_segment)
        print(f"DEBUG_HAAR: Padded data segment to length {n} for even pairing.")

    # Calculate approximation (average) and detail (difference) coefficients
    approximation_coeffs = (data_segment[::2] + data_segment[1::2]) / np.sqrt(2.0)
    detail_coeffs = (data_segment[::2] - data_segment[1::2]) / np.sqrt(2.0)

    return approximation_coeffs, detail_coeffs

def apply_haar_wavelet(spectrum_data, level=1):
    """
    Applies a recursive, manual Haar wavelet transform to a given spectrum.
    This implementation concatenates approximation and detail coefficients from all levels
    into a single feature vector, which might not be standard for DWT analysis.
    
    Args:
        spectrum_data (np.array): The input spectral data.
        level (int): The number of decomposition levels.
        
    Returns:
        np.array: A concatenated array of Haar coefficients.
    """
    if not isinstance(spectrum_data, np.ndarray):
        spectrum_data = np.array(spectrum_data)
        
    if spectrum_data.size == 0:
        print("WARNING_HAAR: Input spectrum is empty, returning empty array.")
        return np.array([])

    current_approximation = spectrum_data.copy()
    all_detail_coefficients = []

    print(f"INFO_HAAR: Starting Haar decomposition for {level} level(s).")
    for l in range(int(level)):
        if current_approximation.size < 2:
            print(f"DEBUG_HAAR: Not enough data for full level {l + 1} decomposition. Stopping.")
            break
        
        approx, detail = _single_level_haar_transform(current_approximation)
        all_detail_coefficients.append(detail)
        current_approximation = approx
        
        print(f"INFO_HAAR: Level {l + 1} processed. Current approx length: {current_approximation.size}, Detail length: {detail.size}")

    # Concatenate the final approximation with all collected detail coefficients.
    # The order might be unusual depending on the intended use case.
    if all_detail_coefficients:
        return np.concatenate([current_approximation] + all_detail_coefficients)
    else:
        return current_approximation # Only approximation if no details were generated

# --- Fourier Transform Feature Extraction (Potentially Irrelevant Components) ---

def extract_fft_features(spectrum_data, num_components=10):
    """
    Extracts Fourier Transform features from the spectrum.
    Returns the magnitude of the first 'num_components' FFT coefficients
    (corresponding to lower frequencies).
    These features were explored as general spectral shape indicators but might not
    directly correlate with specific spectral peaks important for the problem.
    """
    if not isinstance(spectrum_data, np.ndarray):
        spectrum_data = np.array(spectrum_data)

    if spectrum_data.size == 0:
        print("WARNING_FFT: Input spectrum is empty, returning empty array.")
        return np.array([])
    
    # Perform Fast Fourier Transform
    fft_result = fft(spectrum_data)
    
    # Take the magnitude of the positive frequency components
    # The length is N/2 + 1 for real input if N is even, or (N+1)/2 if N is odd
    magnitude_spectrum = np.abs(fft_result[:spectrum_data.size // 2 + 1])
    
    if num_components <= 0:
        print("WARNING_FFT: num_components is non-positive, returning empty array.")
        return np.array([])
    
    # Ensure the feature vector always has 'num_components' length, padding with zeros if necessary.
    # This might hide that fewer components were actually available.
    features = np.zeros(num_components)
    actual_available_components = min(num_components, len(magnitude_spectrum))
    features[:actual_available_components] = magnitude_spectrum[:actual_available_components]

    print(f"INFO_FFT: Extracted {actual_available_components} FFT features (requested {num_components}).")
    print(f"DEBUG_FFT: Max magnitude: {np.max(features):.3f}, Min magnitude: {np.min(features):.3f}")
    
    return features
